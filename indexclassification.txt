                name,
                course: courses[Math.floor(Math.random() * courses.length)],
                present: Math.floor(Math.random() * 20) + 75,
                absent: Math.floor(Math.random() * 5),

                lastAttendance: new Date(Date.now() - Math.random() * 7 * 24 * 60 * 60 * 1000).toLocaleDateString(),

                trend: Math.random() > 0.5 ? 'up' : 'down',
                riskScore: Math.floor(Math.random() * 100)

            predictions = attendanceData.map(s => ({

            currentUser = { role, name: role === 'teacher' ? 'Dr. Sarah Chen' : 'Emma Wilson' };
            document.getElementById('loginScreen').classList.add('hidden');
            document.getElementById('mainApp').classList.remove('hidden');

            document.getElementById('userRole').textContent = role === 'teacher' ? 'Teacher' : 'Student';
            renderSidebar();
            navigate('dashboard');
            lucide.createIcons();
        }

            currentUser = null;

        function navigate(view) {
            currentView = view;
            document.querySelectorAll('.sidebar-link').forEach(el => {
                el.classList.toggle('active', el.dataset.view === view);
                el.classList.toggle('text-slate-400', el.dataset.view !== view);
            });
            
            const titles = {
                dashboard: 'Dashboard',
                attendance: 'Attendance Management',
                'mark-attendance': 'Mark Your Attendance',
                analytics: 'AI Analytics',
                predictions: 'Predictive Insights',
                reports: 'Reports',
                students: 'Student Management'
            };
            document.getElementById('pageTitle').textContent = titles[view] || view;

            renderContent(view);
            lucide.createIcons();
        }

        function renderContent(view) {
            const content = document.getElementById('contentArea');
            
            switch(view) {
                case 'dashboard':
                    content.innerHTML = renderDashboard();
                    break;
                case 'attendance':
                    content.innerHTML = currentUser.role === 'teacher' ? renderAttendanceManager() : renderMyAttendance();
                    break;
                case 'mark-attendance':
                    content.innerHTML = renderMarkAttendance();
                    break;
                case 'analytics':
                    content.innerHTML = renderAnalytics();
                    break;
                case 'predictions':
                    content.innerHTML = renderPredictions();
                    break;
                case 'reports':
                    content.innerHTML = renderReports();
                    break;
                case 'students':
                    content.innerHTML = renderStudents();
                    break;
            }
            
            setTimeout(() => {
                lucide.createIcons();
                initCharts();
            }, 100);
        }
